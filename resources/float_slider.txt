slider_min = -1.0
slider_max = 10.3
slider_step = 0.5


def on_slider_changed(value):
    print(slider_min + ((value / total) * (slider_max - slider_min)))


slider = QtWidgets.QSlider(QtCore.Qt.Horizontal)
value = 4.5

total = int((slider_max - slider_min) / slider_step)
value = int(value / slider_step)

slider.setRange(0, total)
slider.setSingleStep(slider_step)
slider.setValue(value)
slider.valueChanged.connect(on_slider_changed)

slider.show()
